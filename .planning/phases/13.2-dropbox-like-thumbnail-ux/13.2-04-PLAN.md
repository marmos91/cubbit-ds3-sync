```swift
import XCTest
import os
@testable import DS3Lib

/// Memory budget smoke test for the Phase 13.2 fallback render path (D-17, D-18).
///
/// Runs 8 sequential renders on a ~50MB HEIC fixture and asserts that peak resident
/// memory stays under 50MB. This validates the no-size-cap decision (D-17) — the
/// renderer's `Data(contentsOf: .mappedIfSafe)` (Phase 13.1 Finding 2) keeps the
/// original mmap'd, and the 2-slot `ThumbnailFallbackLimiter` bounds peak concurrent
/// memory at ~2 large originals worth (sequential test → peak is 1 original).
///
/// Test is sequential (not concurrent) — Plan 01 already proves the 2-slot bound;
/// this test pins the per-render memory ceiling.
final class ThumbnailFallbackMemorySmokeTests: XCTestCase {

    private var fixtureURL: URL!

    override func setUpWithError() throws {
        fixtureURL = try Self.makeOrLocateLargeHEICFixture()
        let attrs = try FileManager.default.attributesOfItem(atPath: fixtureURL.path)
        let size = (attrs[.size] as? Int64) ?? 0
        XCTAssertGreaterThan(size, 40_000_000, "Fixture must be at least ~40MB to exercise mmap path")
    }

    override func tearDownWithError() throws {
        // If we generated the fixture, clean up. If we located an existing one, leave it.
        if fixtureURL.path.contains(NSTemporaryDirectory()) {
            try? FileManager.default.removeItem(at: fixtureURL)
        }
    }

    func test_eightSequentialRenders_peakMemoryUnder50MB() throws {
        let renderer = ThumbnailRenderer()

        let baseline = Self.residentMB()
        var peak = baseline

        for i in 0..<8 {
            let bytes = try renderer.renderJPEG(from: fixtureURL)
            XCTAssertNotNil(bytes, "Render #\(i) returned nil — fixture may be invalid")
            let current = Self.residentMB()
            peak = max(peak, current)
        }

        let delta = peak - baseline
        // Assert peak DELTA above baseline stays under 50 MB. Absolute resident memory
        // includes test runner overhead which is not what we're measuring.
        XCTAssertLessThan(delta, 50.0,
            "Peak memory delta \(delta) MB exceeds 50 MB ceiling (D-18); baseline=\(baseline) MB peak=\(peak) MB")
    }

    // MARK: - Helpers

    /// Returns current task resident memory in MB. Pattern lifted from
    /// `FileProviderExtension+Thumbnails.swift logMemoryUsage`.
    private static func residentMB() -> Double {
        var info = mach_task_basic_info()
        var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size / MemoryLayout<integer_t>.size)
        let kerr = withUnsafeMutablePointer(to: &info) {
            $0.withMemoryRebound(to: integer_t.self, capacity: Int(count)) {
                task_info(mach_task_self_, task_flavor_t(MACH_TASK_BASIC_INFO), $0, &count)
            }
        }
        guard kerr == KERN_SUCCESS else { return 0 }
        return Double(info.resident_size) / 1_048_576.0
    }

    /// Locates an existing ~50MB HEIC fixture if one exists in the test bundle's
    /// Fixtures directory, OR generates one programmatically into the temp dir.
    private static func makeOrLocateLargeHEICFixture() throws -> URL {
        // Prefer existing committed fixture (Git LFS).
        let bundle = Bundle(for: ThumbnailFallbackMemorySmokeTests.self)
        if let url = bundle.url(forResource: "large_50mb", withExtension: "heic") {
            return url
        }
        // Fallback: programmatically generate a large image.
        // Implementation: compose a 5000x4000 CIImage of solid color noise, encode HEIC.
        // Implement using CIContext.heifRepresentation(of:format:colorSpace:options:).
        let tempURL = URL(fileURLWithPath: NSTemporaryDirectory())
            .appendingPathComponent("large_50mb_\(UUID().uuidString).heic")
        try Self.generateLargeHEIC(at: tempURL, sizeMB: 50)
        return tempURL
    }

    private static func generateLargeHEIC(at url: URL, sizeMB: Int) throws {
        // Use a 5000x4000 noise image — encodes to ~40-60MB HEIC depending on entropy.
        // Implementation deferred to executor — must produce a file >= 40MB on disk
        // that ThumbnailRenderer can decode.
        // Reference: CIImage(generator: .randomGenerator)... cropped, then CIContext heifRepresentation.
        // If generation proves flaky, Git-LFS-track an existing fixture instead and
        // rely on the bundle.url branch above.
        throw XCTSkip("Programmatic 50MB HEIC generation not yet wired — commit Git LFS fixture at DS3DriveProviderTests/Fixtures/large_50mb.heic and re-run.")
    }
}
```
