```swift
import XCTest
import SwiftData
@testable import DS3Lib

/// V4 → V5 migration test (Phase 13.2 D-05, D-06).
///
/// SwiftData "remove field" migrations are an under-trodden path. This test
/// pins the contract: a V4 store with non-default `thumbnailFailCount` values
/// migrates cleanly to V5, all rows survive, all preserved fields retain values.
///
/// V5 simply lacks `thumbnailFailCount` — the type system enforces "field is gone."
final class SchemaV5MigrationTests: XCTestCase {

    func testV4ToV5LightweightMigrationDropsFailCountPreservesRows() throws {
        let storeURL = try Self.makeTempStoreURL()
        defer { try? Self.removeStoreFiles(at: storeURL) }

        let driveId = UUID()
        let s3Key = "folder/photo.heic"

        // Phase 1: seed V4 store with a row carrying non-default thumbnailFailCount.
        do {
            let v4Schema = Schema(versionedSchema: SyncedItemSchemaV4.self)
            let config = ModelConfiguration("MigrationV4V5Test", schema: v4Schema, url: storeURL)
            let container = try ModelContainer(for: v4Schema, configurations: [config])
            let context = ModelContext(container)
            let row = SyncedItemSchemaV4.SyncedItem(
                s3Key: s3Key,
                driveId: driveId,
                size: 12345,
                syncStatus: SyncStatus.synced.rawValue,
                thumbnailStatus: ThumbnailStatus.uploaded.rawValue,
                thumbnailFailCount: 2  // pre-V5 value — must be dropped without crash
            )
            context.insert(row)
            try context.save()
        }

        // Phase 2: re-open with V5 schema + migration plan.
        let v5Schema = Schema(versionedSchema: SyncedItemSchemaV5.self)
        let config = ModelConfiguration("MigrationV4V5Test", schema: v5Schema, url: storeURL)
        let container = try ModelContainer(
            for: v5Schema,
            migrationPlan: SyncedItemMigrationPlan.self,
            configurations: [config]
        )
        let context = ModelContext(container)

        let descriptor = FetchDescriptor<SyncedItemSchemaV5.SyncedItem>()
        let rows = try context.fetch(descriptor)

        XCTAssertEqual(rows.count, 1, "V4 row must survive migration")
        let row = rows[0]
        XCTAssertEqual(row.s3Key, s3Key)
        XCTAssertEqual(row.driveId, driveId)
        XCTAssertEqual(row.size, 12345)
        XCTAssertEqual(row.thumbnailStatus, ThumbnailStatus.uploaded.rawValue)
        // thumbnailFailCount is GONE from the V5 type — type system enforces.
    }

    // MARK: - Helpers (copy from SchemaV4MigrationTests)

    private static func makeTempStoreURL() throws -> URL {
        let dir = FileManager.default.temporaryDirectory
            .appendingPathComponent("schema-v5-test-\(UUID().uuidString)")
        try FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
        return dir.appendingPathComponent("store.sqlite")
    }

    private static func removeStoreFiles(at url: URL) throws {
        for ext in ["", "-shm", "-wal"] {
            let candidate = URL(fileURLWithPath: url.path + ext)
            if FileManager.default.fileExists(atPath: candidate.path) {
                try FileManager.default.removeItem(at: candidate)
            }
        }
    }
}
```

```swift
/// Schema version 5. Drops `thumbnailFailCount` (Phase 13.2 D-05).
///
/// `thumbnailStatus` survives until V6 (Plan 09) because the upload hook
/// writes `.notApplicable` / `.uploaded` to it through the end of Plan 08;
/// D-08 removes those writes in the same plan that ships V6.
public enum SyncedItemSchemaV5: VersionedSchema {
    public nonisolated static let versionIdentifier = Schema.Version(5, 0, 0)
    public static var models: [any PersistentModel.Type] {
        [SyncedItem.self, SyncAnchorRecord.self]
    }
    public typealias SyncAnchorRecord = SyncedItemSchemaV2.SyncAnchorRecord

    @Model
    public final class SyncedItem {
        public var s3Key: String
        public var driveId: UUID
        @Attribute(.unique) public var uniqueKey: String
        public var etag: String?
        public var lastModified: Date?
        public var localFileHash: String?
        public var syncStatus: String
        @Transient public var status: SyncStatus {
            get { SyncStatus(rawValue: syncStatus) ?? .pending }
            set { syncStatus = newValue.rawValue }
        }
        public var parentKey: String?
        public var contentType: String?
        public var size: Int64
        public var isMaterialized: Bool = false
        public var originalKey: String?
        public var thumbnailStatus: String = ThumbnailStatus.pending.rawValue
        @Transient public var thumbnail: ThumbnailStatus {
            get { ThumbnailStatus(rawValue: thumbnailStatus) ?? .pending }
            set { thumbnailStatus = newValue.rawValue }
        }
        // thumbnailFailCount: REMOVED in V5.

        public init(
            s3Key: String,
            driveId: UUID,
            size: Int64 = 0,
            syncStatus: String = SyncStatus.pending.rawValue,
            thumbnailStatus: String = ThumbnailStatus.pending.rawValue
        ) {
            self.s3Key = s3Key
            self.driveId = driveId
            self.uniqueKey = "\(driveId.uuidString):\(s3Key)"
            self.size = size
            self.syncStatus = syncStatus
            self.thumbnailStatus = thumbnailStatus
        }
    }
}
```
