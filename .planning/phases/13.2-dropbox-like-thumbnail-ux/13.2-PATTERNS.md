```swift
// D-19: per-S3-key strike counter, reset only on extension restart.
private var strikeCount: [String: Int] = [:]

// D-19: poison-key set; once added, fallback path returns nil immediately
// for this key until the extension process exits.
private var poisonKeys: Set<String> = []

private static let maxStrikes = 3

/// D-20: returns true if this key is poisoned and the caller should skip
/// download + render entirely (returns nil to perThumbnailCompletionHandler).
func isPoisoned(_ key: String) -> Bool {
    poisonKeys.contains(key)
}

/// D-20: increment counter; on 3rd strike, move into poison set and clear
/// the counter entry. Caller invokes after a render/PUT failure for `key`.
func recordFailure(_ key: String) {
    if poisonKeys.contains(key) { return }
    let next = (strikeCount[key] ?? 0) + 1
    if next >= Self.maxStrikes {
        strikeCount.removeValue(forKey: key)
        poisonKeys.insert(key)
    } else {
        strikeCount[key] = next
    }
}

/// Reset on success — keeps the counter from accumulating for transient
/// errors that eventually resolve (e.g., network blip → retry succeeds).
func recordSuccess(_ key: String) {
    strikeCount.removeValue(forKey: key)
}
```

**Test introspection accessors** (copy lines 113–122 of analog and add `poisonCount` + `strikeCount(for:)` test-only getters).

```swift
/// Fallback render path (D-01 lane 2 + lane 3, D-04, D-12, D-19, D-20, D-24).
///
/// Acquires a slot from the 2-slot `ThumbnailFallbackLimiter`, downloads the
/// original via `s3Lib.downloadS3Item`, renders via `ThumbnailRenderer`,
/// returns the rendered bytes via `perItemHandler`, then fires-and-forgets
/// a `Task.detached` to PUT the same bytes to `.thumbnails/<key>.jpg` and
/// call `signalEnumerator(for: parentContainer)` (D-12).
///
/// Strike + poison handling (D-19, D-20):
/// - Before acquiring a slot, check `limiter.isPoisoned(key)` — if true,
///   return nil immediately, no work done.
/// - On render/PUT failure inside the detached Task, call
///   `limiter.recordFailure(key)`. On success, call `recordSuccess(key)`.
///
/// Pause gate (D-24, THUMB-21): check `SharedData.isDrivePaused(drive.id)`
/// BEFORE acquiring a slot. If paused, return nil — Finder draws default icon.
@Sendable
func consumeThumbnailFallback(
    identifier: NSFileProviderItemIdentifier,
    drive: DS3Drive,
    s3Lib: S3Lib,
    s3Client: any DS3S3ClientProtocol,
    limiter: ThumbnailFallbackLimiter,
    temporaryDirectory: URL,
    domain: NSFileProviderDomain,
    metadataStore: MetadataStore?,
    perItemHandler: PerThumbnailCompletionHandler,
    logger: os.Logger
) async {
    // D-19/D-20: poison check FIRST — skip download + render entirely.
    if await limiter.isPoisoned(identifier.rawValue) {
        perItemHandler(identifier, nil, nil)
        return
    }

    // D-24 / THUMB-21: pause gate before slot acquisition.
    if (try? SharedData.default().isDrivePaused(drive.id)) == true {
        perItemHandler(identifier, nil, nil)
        return
    }

    do {
        try await limiter.acquire()
    } catch is CancellationError {
        perItemHandler(identifier, nil,
            NSError(domain: NSCocoaErrorDomain, code: NSUserCancelledError))
        return
    } catch {
        perItemHandler(identifier, nil, NSFileProviderError(.cannotSynchronize) as NSError)
        return
    }
    defer { Task { await limiter.release() } }

    // Download original + render. On any failure: record strike, return mapped error.
    do {
        let (fileURL, _) = try await s3Lib.downloadS3Item(
            identifier: identifier,
            drive: drive,
            temporaryFolder: temporaryDirectory,
            progress: Progress()
        )
        defer { try? FileManager.default.removeItem(at: fileURL) }

        let renderer = ThumbnailRenderer()
        guard let jpegBytes = try renderer.renderJPEG(from: fileURL) else {
            await limiter.recordFailure(identifier.rawValue)
            perItemHandler(identifier, nil, NSFileProviderError(.noSuchItem) as NSError)
            return
        }

        // D-04: same bytes returned to Finder are PUT to S3 (single render).
        perItemHandler(identifier, jpegBytes, nil)
        await limiter.recordSuccess(identifier.rawValue)

        // Lane 3: fire-and-forget PUT + signalEnumerator. Detached so the
        // per-item handler ordering contract isn't violated by network latency.
        Task.detached(priority: .background) {
            let bucket = drive.syncAnchor.bucket.name
            let thumbKey = S3PathUtils.thumbnailKey(
                forOriginalKey: identifier.rawValue,
                drivePrefix: drive.syncAnchor.prefix
            )
            do {
                try await s3Client.putThumbnail(
                    bucket: bucket, key: thumbKey, jpegData: jpegBytes, sourceETag: nil
                )
                // D-12: signal the parent container so Apple re-enumerates
                // and the next fetchThumbnails hits the now-warm cache.
                if let manager = NSFileProviderManager(for: domain) {
                    let parentKey = S3PathUtils.parentKey(forKey: identifier.rawValue)
                    let parentId = parentKey.map(NSFileProviderItemIdentifier.init)
                        ?? .rootContainer
                    manager.signalEnumerator(for: parentId) { error in
                        if let error {
                            logger.error(
                                "Fallback signalEnumerator failed: \(error.localizedDescription, privacy: .public)"
                            )
                        }
                    }
                }
            } catch {
                logger.error(
                    "Fallback PUT failed for \(thumbKey, privacy: .public): \(DS3S3Client.describeSotoError(error), privacy: .public)"
                )
                // No recordFailure here — the bytes already reached Finder; the user
                // saw a thumbnail. Failed PUT just means next visit re-renders.
            }
        }
    } catch {
        await limiter.recordFailure(identifier.rawValue)
        let mapped = mapThumbnailFetchError(error)
        perItemHandler(identifier, nil, mapped)
    }
}
```

**Pattern source for the detached PUT block** (mirrors `enqueueThumbnailUpload` lines 83–110 in `FileProviderExtension+ThumbnailUploadHook.swift`):

## Shared Patterns

### 1. Actor-isolated FIFO limiter with cancellation

**Source:** `DS3DriveProvider/ThumbnailFetchLimiter.swift` lines 26–122 + `DS3DriveProvider/BucketListingLimiter.swift` lines 14–110

**Apply to:** `ThumbnailFallbackLimiter`

**Rules:**
- Use `actor`, not `class`. Hold `inFlight: Int`, `waiters: [Waiter]`.
- `Waiter` is a private struct with `id: UUID` and `cont: CheckedContinuation<Void, any Error>`.
- Wrap `withCheckedThrowingContinuation` inside `withTaskCancellationHandler { ... } onCancel: { Task { [weak self] in await self?.cancelWaiter(id: ...) } }`.
- Post-resume guard: `if Task.isCancelled { release(); throw CancellationError() }`.
- Hand-off in `release()`: `let next = waiters.removeFirst(); next.cont.resume()` — do NOT decrement `inFlight` on hand-off (slot stays counted).
- Ownership note (lines 8–16 of analog): the limiter is held as `private let` on the long-lived `FileProviderExtension`; deinit-while-suspended is structurally impossible.

### 2. Fire-and-forget detached Task with sendable capture

**Source:** `DS3DriveProvider/FileProviderExtension+ThumbnailUploadHook.swift` lines 75–110, `DS3DriveProvider/FileProviderExtension+ThumbnailCascade.swift` lines 50–76, 119–181

**Apply to:** Fallback PUT in `consumeThumbnailFallback`; post-PUT `signalEnumerator` in upload hook

**Rules (verbatim from `+ThumbnailUploadHook.swift` line 27 and `+ThumbnailCascade.swift` line 27):**
- "**NEVER capture `self` inside the detached Task** — pass sendable locals only: `s3Client` (Sendable), `metadataStore` (`@ModelActor`), `drive` (Sendable struct), strings, and the logger."
- Free function (NOT a method on `FileProviderExtension`); the FP extension subclass is non-Sendable.
- Errors logged + swallowed inside the detached Task — they NEVER propagate to the user-visible contract.
- Use `Task.detached(priority: .background)` for the outer wrapper.
- If parameter count ≥ 5, bundle into a Sendable struct (mirror `ThumbnailUploadHookContext`).

### 3. Soto error logging via `describeSotoError`

**Source:** Phase 13.1 sweep — every catch in `FileProviderExtension+Create.swift`, `FileProviderExtension+Modify.swift`, `FileProviderExtension+Thumbnails.swift` uses `DS3S3Client.describeSotoError(error)`.

**Apply to:** All new catches in `consumeThumbnailFallback` and the upload-hook signal block.

```swift
} catch {
    logger.error(
        "Fallback PUT failed for \(thumbKey, privacy: .public): \(DS3S3Client.describeSotoError(error), privacy: .public)"
    )
}
```

NEVER use `error.localizedDescription` — Soto errors return generic strings that hide the S3 error code (per MEMORY.md and CLAUDE.md).

### 4. OSLog `privacy: .public` on dynamic strings

**Source:** Codebase-wide (CLAUDE.md mandate; MEMORY.md "Swift 6 Concurrency").

**Apply to:** Every new log statement in Phase 13.2.

```swift
logger.info("fetchThumbnails fallback: rendering \(originalKey, privacy: .public)")
```

### 5. Pause gate per-handler

**Source:** `FileProviderExtension+Thumbnails.swift` lines 233–240 (existing `fetchThumbnails`), `FileProviderExtension.swift` line 385 (`isDrivePaused`).

**Apply to:** `consumeThumbnailFallback` BEFORE acquiring a slot (D-24, THUMB-21).

```swift
if (try? SharedData.default().isDrivePaused(drive.id)) == true {
    perItemHandler(identifier, nil, nil)  // Finder draws default icon; no error surface.
    return
}
```

### 6. NSFileProviderErrorDomain / NSCocoaErrorDomain only

**Source:** `FileProviderExtension+ThumbnailConsume.swift` lines 113–172 (`mapThumbnailFetchError`); MEMORY.md.

**Apply to:** Every error returned to `perItemHandler` from the fallback path. Reuse `mapThumbnailFetchError(error)` directly — it already covers the cases we need (auth, throttling, network, unknown).

### 7. SwiftData VersionedSchema migration test

**Source:** `DS3Lib/Tests/DS3LibTests/SchemaV4MigrationTests.swift`

**Apply to:** `SchemaV5MigrationTests.swift`, `SchemaV6MigrationTests.swift`

**Rules:**
- Use `Schema(versionedSchema: SyncedItemSchemaVN.self)` to seed.
- Use `ModelConfiguration(name, schema:, url:)` with a temp file path (`makeTempStoreURL` helper).
- Re-open with the next-version schema + `SyncedItemMigrationPlan` and fetch — survival is the assertion.
- Assert default values land per-row: `XCTAssertEqual(row.thumbnailFailCount, 0)` on V3→V4 migration; on V4→V5 migration, the field doesn't exist on the V5 model — type-system enforcement is the test.
