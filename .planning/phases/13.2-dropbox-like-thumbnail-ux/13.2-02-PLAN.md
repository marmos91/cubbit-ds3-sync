```swift
/// Fallback render path for `fetchThumbnails` cache miss (Phase 13.2, D-01..D-04, D-12, D-19, D-20, D-24).
///
/// Sequence:
///   1. Check `limiter.isPoisoned(key)` → if true, return nil (no work).
///   2. Check `SharedData.isDrivePaused(drive.id)` → if true, return nil (THUMB-21).
///   3. Acquire a slot on the 2-slot `ThumbnailFallbackLimiter`.
///   4. Download original via `s3Lib.downloadS3Item` to temporaryDirectory.
///   5. Render via `ThumbnailRenderer.renderJPEG(from: fileURL)`.
///   6. Return rendered bytes to Finder via `perItemHandler` (D-01 lane 2, D-04).
///   7. Fire-and-forget `Task.detached` that PUTs the same bytes to .thumbnails/<key>.jpg
///      (D-01 lane 3, D-04) and calls `signalParentContainer(parentId)` (D-12).
///
/// **Sendable rules (Pitfall 1, +ThumbnailUploadHook.swift line 27):**
///   - Free function, NOT a method on FileProviderExtension (subclass is non-Sendable).
///   - Never captures `self`.
///   - All parameters are Sendable: `s3Client` (Sendable protocol), `s3Lib` (struct),
///     `drive` (Sendable), `limiter` (actor), URL (Sendable), `domain` (Sendable),
///     closures explicitly @Sendable.
///
/// **Error handling (THUMB-13):** all errors mapped via `mapThumbnailFetchError`
/// to NSFileProviderErrorDomain or NSCocoaErrorDomain. Detached PUT errors logged
/// via `DS3S3Client.describeSotoError` (NEVER `.localizedDescription`).
@Sendable
func consumeThumbnailFallback(
    identifier: NSFileProviderItemIdentifier,
    drive: DS3Drive,
    s3Lib: S3Lib,
    s3Client: any DS3S3ClientProtocol,
    limiter: ThumbnailFallbackLimiter,
    temporaryDirectory: URL,
    signalParentContainer: @escaping @Sendable (NSFileProviderItemIdentifier) -> Void,
    perItemHandler: @escaping PerThumbnailCompletionHandler,
    logger: os.Logger
) async {
    let key = identifier.rawValue

    // D-19/D-20: poison check FIRST — skip everything for poisoned keys.
    if await limiter.isPoisoned(key) {
        logger.info("Fallback: skipping poisoned key \(key, privacy: .public)")
        perItemHandler(identifier, nil, nil)
        return
    }

    // D-24 / THUMB-21: pause gate BEFORE acquiring a slot.
    if (try? SharedData.default().isDrivePaused(drive.id)) == true {
        logger.info("Fallback: drive paused, skipping \(key, privacy: .public)")
        perItemHandler(identifier, nil, nil)
        return
    }

    // Acquire 2-slot limiter (D-02). Cancellation maps to NSUserCancelledError.
    do {
        try await limiter.acquire()
    } catch is CancellationError {
        perItemHandler(identifier, nil,
                       NSError(domain: NSCocoaErrorDomain, code: NSUserCancelledError))
        return
    } catch {
        perItemHandler(identifier, nil, mapThumbnailFetchError(error))
        return
    }
    defer { Task { await limiter.release() } }

    do {
        // Lane 2 step 1: download original.
        let (fileURL, _) = try await s3Lib.downloadS3Item(
            identifier: identifier,
            drive: drive,
            temporaryFolder: temporaryDirectory,
            progress: Progress()
        )
        defer { try? FileManager.default.removeItem(at: fileURL) }

        // Lane 2 step 2: render. Renderer returns nil for unsupported formats.
        let renderer = ThumbnailRenderer()
        guard let jpegBytes = try renderer.renderJPEG(from: fileURL) else {
            await limiter.recordFailure(key)
            perItemHandler(identifier, nil, NSFileProviderError(.noSuchItem) as NSError)
            return
        }

        // D-04: same bytes to Finder AND to S3 — single render.
        perItemHandler(identifier, jpegBytes, nil)
        await limiter.recordSuccess(key)

        // Lane 3: fire-and-forget PUT + signalEnumerator. Detached so the per-item
        // handler ordering contract isn't violated by network latency.
        let bucket = drive.syncAnchor.bucket.name
        let drivePrefix = drive.syncAnchor.prefix
        let parentKey = S3PathUtils.parentKey(forKey: key)
        let parentId: NSFileProviderItemIdentifier =
            parentKey.map(NSFileProviderItemIdentifier.init) ?? .rootContainer

        Task.detached(priority: .background) {
            let thumbKey = S3PathUtils.thumbnailKey(
                forOriginalKey: key, drivePrefix: drivePrefix
            )
            do {
                try await s3Client.putThumbnail(
                    bucket: bucket, key: thumbKey,
                    jpegData: jpegBytes, sourceETag: nil
                )
                logger.info("Fallback: PUT succeeded for \(thumbKey, privacy: .public)")
                // D-12: signal parent so Apple re-enumerates and next visit hits cache.
                signalParentContainer(parentId)
            } catch {
                logger.error(
                    "Fallback PUT failed for \(thumbKey, privacy: .public): \(DS3S3Client.describeSotoError(error), privacy: .public)"
                )
                // No recordFailure here — the user already saw a thumbnail. Failed
                // PUT just means next visit re-renders.
            }
        }
    } catch {
        await limiter.recordFailure(key)
        logger.error(
            "Fallback render failed for \(key, privacy: .public): \(DS3S3Client.describeSotoError(error), privacy: .public)"
        )
        perItemHandler(identifier, nil, mapThumbnailFetchError(error))
    }
}
```

| Boundary | Description |
|----------|-------------|
| Apple FP framework → fallback path | `fetchThumbnails` arrives with attacker-influenced item identifiers (S3 keys); the fallback path downloads originals and renders them |
| DS3DriveProvider → S3 (PUT thumbnails) | Detached Task PUT writes thumbnails; failures must not corrupt user's view |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13.2-04 | Information Disclosure | Custom error type leaks across FP boundary | mitigate | All errors flow through `mapThumbnailFetchError` → NSFileProviderErrorDomain or NSCocoaErrorDomain only; verified by acceptance criterion (no `localizedDescription` on Soto errors, only `describeSotoError`) |
| T-13.2-05 | DoS (disk fill) | Temp file leak from fallback render | mitigate | `defer { try? FileManager.default.removeItem(at: fileURL) }` immediately after download — same pattern as the dying `ThumbnailBackfillCoordinator.processItem` |
| T-13.2-06 | DoS (memory) | Two simultaneous large originals exceed jetsam | mitigate | 2-slot limiter (Plan 01) + `Data(contentsOf: .mappedIfSafe)` in renderer (Phase 13.1 Finding 2 fix) keep peak resident memory bounded; Plan 04 ships memory smoke test |
| T-13.2-07 | Tampering | Race between fallback PUT and rename cascade | accept | Window is tiny (cascade fires on user action, not poll); Phase 13 D-22 cascade copy preserves metadata; documented as known-tiny window in PATTERNS.md |
