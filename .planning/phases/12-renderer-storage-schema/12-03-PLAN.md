    ```swift
    import Foundation
    import SotoS3
    import NIOCore

    public extension DS3S3ClientProtocol {

        /// Uploads a thumbnail as a single-part PUT with staleness + format-version metadata.
        /// - Parameter data: must be < `DefaultSettings.Thumbnail.maxSinglePartBytes` (500 KB) — traps otherwise.
        /// - Parameter sourceETag: normalized ETag of the source object (stored in `x-amz-meta-source-etag`).
        /// - Returns: normalized ETag of the uploaded thumbnail.
        func putThumbnail(
            bucket: String,
            key: String,
            data: Data,
            sourceETag: String
        ) async throws -> String {
            precondition(
                data.count < DefaultSettings.Thumbnail.maxSinglePartBytes,
                "Thumbnails must be <500 KB single-part. Got \(data.count) bytes for key \(key)."
            )

            let metadata: [String: String] = [
                DefaultSettings.Thumbnail.sourceETagMetadataKey: sourceETag,
                DefaultSettings.Thumbnail.formatVersionMetadataKey: "\(DefaultSettings.Thumbnail.formatVersion)"
            ]

            let etag = try await putObjectData(
                bucket: bucket, key: key, data: data, metadata: metadata
            )
            // putObjectData already normalizes; guard against nil (never expected in practice)
            return etag ?? ""
        }

        /// Fetches thumbnail bytes. Returns nil on 404; throws on network / auth / 5xx.
        func getThumbnailBytes(
            bucket: String,
            key: String
        ) async throws -> Data? {
            do {
                // Use the smallest surface on the protocol that returns Data.
                // If the protocol already exposes `getObjectData` use it; otherwise
                // call `getObject(bucket:key:toFile:)` to a temp URL and read Data —
                // the existing DS3S3Client+Transfers surface provides one of these.
                return try await getObjectData(bucket: bucket, key: key)
            } catch {
                if DS3S3Client.isNotFoundError(error) { return nil }
                throw error
            }
        }

        /// Deletes a thumbnail. Silent on 404 (D-14). Throws on network / auth / 5xx.
        func deleteThumbnail(
            bucket: String,
            key: String
        ) async throws {
            do {
                try await deleteObject(bucket: bucket, key: key)
            } catch {
                if DS3S3Client.isNotFoundError(error) { return }
                throw error
            }
        }
    }
    ```
