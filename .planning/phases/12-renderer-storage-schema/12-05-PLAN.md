    ```swift
    import Foundation
    import os

    /// Orchestrates opportunistic thumbnail generation for a single drive.
    ///
    /// - Phase 12: scaffold-only. Ships with a runnable `runBatch` but no production caller.
    /// - Phase 13: BFS backfill hook calls `runBatch(maxItems: 5)` per pass.
    /// - Phase 14: iOS overnight task calls `runBatch` in a loop until expiration.
    ///
    /// The actor shell is CROSS-PLATFORM (compiles on iOS + macOS) per D-29.
    /// Only the render step inside `runBatch` is `#if os(macOS)` — iOS render path is
    /// Phase 14's responsibility.
    public actor ThumbnailBackfillCoordinator {

        public struct BatchResult: Sendable {
            public let processed: Int
            public let succeeded: Int
            public let skipped: Int    // .notApplicable transitions
            public let failed: Int     // .failed transitions
        }

        private let metadataStore: MetadataStore
        private let s3Client: any DS3S3ClientProtocol
        private let drive: DS3Drive
        private let logger = Logger(subsystem: "io.cubbit.DS3Drive", category: "thumbnail.backfill")

        public init(
            metadataStore: MetadataStore,
            s3Client: any DS3S3ClientProtocol,
            drive: DS3Drive
        ) {
            self.metadataStore = metadataStore
            self.s3Client = s3Client
            self.drive = drive
        }

        /// Runs one backfill pass, processing up to `maxItems` pending thumbnails.
        /// - Returns: `BatchResult` counts for the pass.
        /// - Note: `maxItems` is a FETCH bound — the raster allow-list filter in
        ///   `fetchPendingThumbnails` may return fewer items than requested (Pitfall 5
        ///   in 12-RESEARCH.md). Callers MUST NOT infer end-of-queue from `processed < maxItems`.
        public func runBatch(maxItems: Int) async throws -> BatchResult {
            let pending = try await metadataStore.fetchPendingThumbnails(
                driveId: drive.id, limit: maxItems
            )

            if pending.isEmpty {
                return BatchResult(processed: 0, succeeded: 0, skipped: 0, failed: 0)
            }

            var succeeded = 0
            var skipped = 0
            var failed = 0

            // Temp-file cleanup guard — Pitfall 9. Matches FileProviderExtension+Thumbnails.swift:219-224.
            var downloadedFiles: [URL] = []
            defer {
                for file in downloadedFiles {
                    try? FileManager.default.removeItem(at: file)
                }
            }

            for item in pending {
                do {
                    let tempURL = FileManager.default.temporaryDirectory
                        .appendingPathComponent(UUID().uuidString)
                    downloadedFiles.append(tempURL)

                    _ = try await s3Client.getObject(
                        bucket: drive.syncAnchor.bucket,
                        key: item.s3Key,
                        toFile: tempURL,
                        onProgress: nil
                    )

                    #if os(macOS)
                        let renderer = ThumbnailRenderer()   // lazy construct per D-30, DefaultSettings knobs per D-01
                        if let thumbnailData = renderer.renderJPEG(from: tempURL) {
                            let thumbKey = S3PathUtils.thumbnailKey(
                                forOriginalKey: item.s3Key,
                                drivePrefix: drive.syncAnchor.prefix
                            )
                            _ = try await s3Client.putThumbnail(
                                bucket: drive.syncAnchor.bucket,
                                key: thumbKey,
                                data: thumbnailData,
                                sourceETag: item.etag ?? ""
                            )
                            try await metadataStore.setThumbnailStatus(
                                s3Key: item.s3Key, driveId: drive.id, status: .uploaded
                            )
                            succeeded += 1
                        } else {
                            // Render returned nil — distinguish unsupported vs. supported-but-failed
                            // by re-running the allow-list check via UTI. Minimum policy per D-32;
                            // retry budgets / negative cache / 3-strike are Phase 13 (THUMB-20).
                            let ext = (item.s3Key as NSString).pathExtension.lowercased()
                            let rasterExts: Set<String> = [
                                "jpg","jpeg","png","heic","heif","webp","gif","tiff","tif"
                            ]
                            if rasterExts.contains(ext) {
                                try await metadataStore.setThumbnailStatus(
                                    s3Key: item.s3Key, driveId: drive.id, status: .failed
                                )
                                failed += 1
                            } else {
                                try await metadataStore.setThumbnailStatus(
                                    s3Key: item.s3Key, driveId: drive.id, status: .notApplicable
                                )
                                skipped += 1
                            }
                        }
                    #else
                        // iOS render path is Phase 14's responsibility. In Phase 12, no iOS caller
                        // exists — if this branch ever runs, mark the item .failed and move on.
                        // (Formally undefined per D-29 — Phase 14 extends this branch.)
                        try await metadataStore.setThumbnailStatus(
                            s3Key: item.s3Key, driveId: drive.id, status: .failed
                        )
                        failed += 1
                    #endif
                } catch {
                    logger.error("Thumbnail backfill item failed: \(item.s3Key, privacy: .public) — \(String(describing: error), privacy: .public)")
                    try? await metadataStore.setThumbnailStatus(
                        s3Key: item.s3Key, driveId: drive.id, status: .failed
                    )
                    failed += 1
                }
            }

            return BatchResult(
                processed: pending.count,
                succeeded: succeeded,
                skipped: skipped,
                failed: failed
            )
        }
    }
    ```
