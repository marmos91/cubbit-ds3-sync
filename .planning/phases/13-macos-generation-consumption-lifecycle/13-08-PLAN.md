    **Step A — Create `DS3DriveProvider/FileProviderExtension+ThumbnailCascade.swift`** (new file; helpers live here per Pitfall 12 file-length watch):

    ```swift
    import Foundation
    import DS3Lib

    extension FileProviderExtension {

        #if os(macOS)
        /// Fire-and-forget delete cascade after the original-file S3 delete succeeds.
        /// Per Phase 13 D-21. 404 silent (Phase 12 D-14). Failures fall to orphan sweep.
        func enqueueDeleteCascade(originalKey: String) {
            guard S3PathUtils.isRasterExtension((originalKey as NSString).pathExtension) else { return }
            let s3ClientCopy: any DS3S3ClientProtocol = s3Lib.dsClient
            let driveCopy = drive
            let logger = self.logger
            Task.detached {
                let thumbKey = S3PathUtils.thumbnailKey(forOriginalKey: originalKey, drivePrefix: driveCopy.syncAnchor.prefix)
                do {
                    try await s3ClientCopy.deleteThumbnail(bucket: driveCopy.syncAnchor.bucket.name, key: thumbKey)
                } catch {
                    logger.error("Delete cascade failed for \(thumbKey, privacy: .public): \(error.localizedDescription, privacy: .public)")
                }
            }
        }

        /// Fire-and-forget rename/move cascade after the original-file rename completes.
        /// Server-side copy + delete-old. On copy failure, marks new key .pending so backfill regenerates.
        /// Per Phase 13 D-22, D-24.
        func enqueueRenameCascade(oldOriginalKey: String, newOriginalKey: String) {
            guard S3PathUtils.isRasterExtension((newOriginalKey as NSString).pathExtension) else { return }
            let s3ClientCopy: any DS3S3ClientProtocol = s3Lib.dsClient
            let metadataStoreCopy = metadataStore
            let driveCopy = drive
            let logger = self.logger
            Task.detached {
                let prefix = driveCopy.syncAnchor.prefix
                let bucket = driveCopy.syncAnchor.bucket.name
                let oldThumb = S3PathUtils.thumbnailKey(forOriginalKey: oldOriginalKey, drivePrefix: prefix)
                let newThumb = S3PathUtils.thumbnailKey(forOriginalKey: newOriginalKey, drivePrefix: prefix)
                do {
                    try await s3ClientCopy.copyThumbnail(bucket: bucket, fromKey: oldThumb, toKey: newThumb)
                } catch {
                    logger.error("Rename cascade copy failed; marking new key .pending: \(newThumb, privacy: .public): \(error.localizedDescription, privacy: .public)")
                    try? await metadataStoreCopy.setThumbnailStatus(s3Key: newOriginalKey, driveId: driveCopy.id, status: .pending)
                    return
                }
                do {
                    try await s3ClientCopy.deleteThumbnail(bucket: bucket, key: oldThumb)
                } catch {
                    logger.warning("Rename cascade delete-old failed (orphan sweep will reclaim): \(oldThumb, privacy: .public)")
                }
            }
        }
        #endif
    }
    ```
