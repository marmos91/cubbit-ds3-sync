    2. **S3PathUtils.swift** — After the `trashParentKey` method (line 96), add four new static methods mirroring the trash helper set. The critical difference from trash is the per-folder `.thumbnails/` placement + `.jpg` append rule:

       ```swift
       // MARK: - Thumbnail Helpers

       /// Computes the thumbnails prefix for a drive (e.g., "prefix/.thumbnails/").
       public static func thumbnailsPrefix(forDrivePrefix drivePrefix: String?) -> String {
           (drivePrefix ?? "") + DefaultSettings.S3.thumbnailsPrefix
       }

       /// Returns true if the key lives inside any `.thumbnails/` segment.
       public static func isThumbnailKey(_ key: String, drivePrefix: String?) -> Bool {
           // Check if any path component is ".thumbnails/" — handles both root and per-folder placement.
           let prefix = drivePrefix ?? ""
           let relativePath = key.hasPrefix(prefix) ? String(key.dropFirst(prefix.count)) : key
           return relativePath.contains(DefaultSettings.S3.thumbnailsPrefix)
       }

       /// Computes the thumbnail key for an original key.
       /// Rule (THUMB-03 / D-15): `.thumbnails/` folder is placed in the SAME directory as the original,
       /// and `.jpg` is appended to the full filename (not substituted).
       /// Example: "prefix/photos/a.heic" with drivePrefix "prefix/" -> "prefix/photos/.thumbnails/a.heic.jpg"
       public static func thumbnailKey(forOriginalKey key: String, drivePrefix: String?) -> String {
           let delimiter = DefaultSettings.S3.delimiter
           let components = key.split(separator: delimiter, omittingEmptySubsequences: false)
           guard components.count > 0 else { return key }

           let filename = String(components.last!)
           let parentComponents = components.dropLast()
           let parent = parentComponents.isEmpty ? "" : parentComponents.joined(separator: String(delimiter)) + String(delimiter)

           return parent + DefaultSettings.S3.thumbnailsPrefix + filename + ".jpg"
       }

       /// Derives the original key from a thumbnail key.
       /// Reverses thumbnailKey: strips the `.thumbnails/` segment from the parent path
       /// and removes the trailing `.jpg` from the filename.
       public static func originalKey(fromThumbnailKey key: String, drivePrefix: String?) -> String {
           let thumbPrefix = DefaultSettings.S3.thumbnailsPrefix
           // Find the last occurrence of ".thumbnails/" in the key
           guard let range = key.range(of: thumbPrefix, options: .backwards) else { return key }
           let parent = String(key[key.startIndex..<range.lowerBound])
           var filename = String(key[range.upperBound...])
           // Strip the trailing ".jpg" that was appended
           if filename.hasSuffix(".jpg") {
               filename = String(filename.dropLast(4))
           }
           return parent + filename
       }
       ```

       IMPORTANT: The `isThumbnailKey` implementation must handle nested cases like `prefix/sub/.thumbnails/a.jpg` where `.thumbnails/` is NOT at the drive root but deeper in the path hierarchy. The `contains` approach handles this. Test carefully against the test table.

       import Foundation

       /// Centralized filter that determines whether an S3 key represents user-visible content.
       /// Routes through S3PathUtils predicates for .trash/ and .thumbnails/ hidden prefixes.
       /// All ListObjectsV2 consumers MUST use this filter before surfacing keys to observers,

       public enum S3KeyFilter {

           /// - Parameters:
           ///   - key: The S3 object key

           /// - Returns: true if the key is user-facing content
           public static func isUserVisible(key: String, drivePrefix: String?) -> Bool {
