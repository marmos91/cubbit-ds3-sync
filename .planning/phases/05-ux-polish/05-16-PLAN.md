    ```swift
    import Foundation

    actor BucketListingLimiter {
        static let shared = BucketListingLimiter()
        private let maxConcurrent: Int
        private var inFlight = 0
        private var waiters: [CheckedContinuation<Void, Never>] = []

        init(maxConcurrent: Int = 4) { self.maxConcurrent = maxConcurrent }

        func acquire() async {
            if inFlight < maxConcurrent {
                inFlight += 1
                return
            }
            await withCheckedContinuation { cont in waiters.append(cont) }
            inFlight += 1
        }

        func release() {
            inFlight -= 1
            if !waiters.isEmpty {
                let next = waiters.removeFirst()
                next.resume()
            }
        }

        func withLimit<T>(_ body: () async throws -> T) async rethrows -> T {
            await acquire()
            defer { Task { await release() } }
            return try await body()
        }
    }
    ```
    NOTE: Verify the `withLimit` release path is correct for both success and throw. Adjust if Swift concurrency trips on the `defer { Task }` pattern — use a manual `do { ... release(); return } catch { release(); throw }` if needed.

    Step 2 — In `DS3DriveProvider/S3Lib.swift`, add a retry helper. Use the exact AWSResponseError matching used elsewhere in the extension (Plan 05-07 added retry for HTTP/2 StreamClosed — follow the same pattern):
    ```swift
    func listWithRetries(
        _ input: ListObjectsV2Request,
        using s3: S3,
        maxAttempts: Int = 5,
        logger: Logger
    ) async throws -> ListObjectsV2Output {
        var attempt = 0
        while true {
            do {
                return try await BucketListingLimiter.shared.withLimit {
                    try await s3.listObjectsV2(input)
                }
            } catch let error as AWSResponseError where error.errorCode == "SlowDown" || error.statusCode == 503 {
                attempt += 1
                if attempt >= maxAttempts {
                    logger.error("S3 SlowDown: exhausted \(maxAttempts) retries on prefix \(input.prefix ?? "nil", privacy: .public)")
                    throw error
                }
                let baseDelay = 0.5 * pow(2.0, Double(attempt))
                let jitter = Double.random(in: 0...0.5)
                let totalDelay = baseDelay + jitter
                logger.warning("S3 SlowDown on prefix \(input.prefix ?? "nil", privacy: .public), retry \(attempt)/\(maxAttempts) after \(totalDelay)s")
                try await Task.sleep(for: .seconds(totalDelay))
            }
        }
    }
    ```
    Delay schedule: 1.0-1.5s, 2.0-2.5s, 4.0-4.5s, 8.0-8.5s, 16.0-16.5s (attempt=1..5).
