<objective>
Implement the CI parity gate (D-25) that catches schema drift between Rust serde and Swift Codable BEFORE upgrade ships. Without this gate, a Rust developer renaming a field in `ds3-models::Account` could silently break `accountSession.json` decoding for every existing user on app upgrade.

The gate has two halves: (1) Rust side reads canonical production-shaped JSON fixtures and round-trips through serde; (2) Swift side reads the same bytes through `Bundle.module.url(forResource:)` and asserts Codable decode produces equivalent field values. Both run in CI.

Output: Four committed JSON fixtures, extended Rust tests, a new SchemaParityTests, and a CI workflow that runs `cargo test --tests` (currently only `--lib`).
</objective>

    import XCTest
    @testable import DS3Lib

    final class SchemaParityTests: XCTestCase {
        // Helper — adapt to chosen resource-loading path from Package.swift edit above.
        private func loadFixture(_ name: String) throws -> Data {
            guard let url = Bundle.module.url(forResource: name, withExtension: "json")
            else { XCTFail("fixture not found: \(name).json"); throw XCTSkip("missing fixture") }
            return try Data(contentsOf: url)
        }

        func testDrivesFixtureDecodes() throws {
            let data = try loadFixture("drives_v4")
            let drives = try JSONDecoder().decode([DS3Drive].self, from: data)
            XCTAssertFalse(drives.isEmpty)
            XCTAssertEqual(drives[0].syncAnchor.bucket.name, "test-bucket-1")
        }

        func testCredentialsFixtureDecodes() throws {
            let data = try loadFixture("credentials_v1")
            _ = try JSONDecoder().decode([DS3ApiKey].self, from: data)
        }

        func testAccountSessionFixtureDecodes() throws {
            let data = try loadFixture("accountSession_v1")
            let s = try JSONDecoder().decode(AccountSession.self, from: data)
            XCTAssertFalse(s.refreshToken.isEmpty)
        }

        func testAccountFixtureDecodes() throws {
            let data = try loadFixture("account_v1")
            let a = try JSONDecoder().decode(Account.self, from: data)
            XCTAssertFalse(a.endpointGateway.isEmpty)  // field name verifies camelCase mapping
        }
    }
